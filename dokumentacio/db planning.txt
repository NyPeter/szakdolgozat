User:
    -id: id
    -name: string
    -email: string
    -password: string
        -games: int
        -wins: int
    -avatarURL: string
    -isAdmin: bool
    -suspendedUntil: null or DateTime
    -mutedUntil: null or DateTime

Game:
    -id: id
    -status: enum(waiting, started, finished)
    -ownerID: int - foreign key refers to User.id
    -winnerID: int - foreign key refers to User.id

Chat:
    -id: id
    -message: text
    -userID: int - foreign key refers to User.id
    -gameID: int - foreign key refers to Game.id

GameUser:
    -id: id
    -isSurrendered: bool
    -gameID: int - foreign key refers to Game.id
    -userID: int - foreign key refers to User.id

relations:
    User - Game: N - N
    Chat - Game: N - 1
    Chat - User: N - 1
